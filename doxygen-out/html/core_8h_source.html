51 /* Framing scratch for cc3501e_request() (#740 scope note): these were
52 * ALREADY per-instance fields before this change (never the
53 * function-local `static` pattern the scan/event buffers below used
54 * to have) and their lifetime was already bounded to a single
55 * cc3501e_request() call -- filled, consumed by the caller-supplied
56 * rx_buf memcpy, and never read back across calls -- so they carry
57 * none of the #740 aliasing risk and needed no change here. */
58 uint8_t rx_scratch[ALP_CC3501E_HEADER_BYTES + ALP_CC3501E_MAX_PAYLOAD];
59 uint8_t tx_scratch[ALP_CC3501E_HEADER_BYTES + ALP_CC3501E_MAX_PAYLOAD];
60 /* Per-context decode scratch for the scan/event helpers (issue #740).
61 * Each of these used to be a function-local `static` buffer in
62 * cc3501e_wifi.c / cc3501e_ble.c / cc3501e_events.c -- process-global
63 * storage shared by EVERY cc3501e_t instance, so two contexts (or a
64 * caller re-entering the same context, e.g. from inside the event
65 * callback) could alias and corrupt each other's in-flight decode.
66 * Moving the storage in here makes it per-instance, matching
67 * rx_scratch/tx_scratch above; the *_busy flags make same-instance
68 * reentrancy an explicit ALP_ERR_BUSY instead of silent aliasing (see
69 * cc3501e_wifi_scan / cc3501e_ble_scan / cc3501e_poll_events).
70 *
71 * Three SEPARATE ALP_CC3501E_MAX_PAYLOAD (512 B) buffers -- not one
72 * shared "radio scratch" reused across all three helpers -- because
73 * cc3501e_poll_events() is meant to be polled from a low-rate app
74 * thread (or the CONFIG_ALP_SDK_CC3501E_EVENT_IRQ workqueue)
75 * regardless of whatever else the app is doing with the SAME ctx, so
76 * an app that calls e.g. cc3501e_wifi_scan() and polls events from a
77 * different call site must not have one invalidate the other's
78 * decode; and collapsing wifi_scan_buf/ble_scan_buf into one shared
79 * buffer would silently reintroduce the exact same-ctx aliasing this
80 * struct exists to remove the moment an app pipelines a Wi-Fi scan
81 * and a BLE scan close together. This grows sizeof(cc3501e_t) from
82 * 1088 to 2632 bytes (measured, GCC 13.3 host build) -- accepted
83 * because the driver context is a small, fixed number of
84 * long-lived, typically-static allocations per module (one CC3501E
85 * per E1M-AEN board), not a per-connection/per-packet object; halving
86 * the number of scratch buffers would only save ~1.5 KB while giving
87 * up the correctness property #740 exists for. */
88 uint8_t wifi_scan_buf[ALP_CC3501E_MAX_PAYLOAD];
89 uint8_t ble_scan_buf[ALP_CC3501E_MAX_PAYLOAD];
90 uint8_t evt_buf[ALP_CC3501E_MAX_PAYLOAD];
91 bool wifi_scan_busy;
92 bool ble_scan_busy;
93 bool evt_busy;
94 /* Transport-transaction lock (issue #1116): serialises the whole
95 * cc3501e_request() 4-phase exchange (and therefore tx_scratch /
96 * rx_scratch above) across every caller sharing this ctx -- the
97 * Wi-Fi / BLE / GPIO-proxy backends, the console companion, and the
98 * OTA path. A plain flag guarded by compiler-builtin atomics
99 * (__atomic_* in cc3501e_core.c), not an OS mutex: this driver core
100 * is OS-agnostic (chips/cc3501e/cc3501e_core.c links into the
101 * Zephyr module AND the plain-CMake / Yocto libalp_chips.a build,
102 * per CMakeLists.txt's ALP_SDK_CHIP_LIST comment), so it cannot call
103 * k_mutex_*. Same rationale as src/common/alp_slot_claim.h's
104 * lock-free slot claim. Never touch directly -- go through
105 * cc3501e_request(). */
106 bool request_lock;
107};
