92/* SDIO 74LVC157 multiplexer (M.2 E-key SDIO vs microSD card slot).
93 *
94 * Two 74LVC157 quad 2:1 muxes (U38 for data lines, U39 for CLK/CMD/RST)
95 * pick which device drives the SoM's single SDIO bus. Control pins:
96 *
97 * /E (active-low enable) = E1M IO20 -- both U38 and U39 share /E
98 * S (select) = E1M IO21
99 *
100 * Per the 74LVC157 truth table:
101 * /E = 0, S = 0 -> M.2 E-key SDIO routed to SoM
102 * /E = 0, S = 1 -> microSD card slot routed to SoM
103 * /E = 1 -> outputs Hi-Z (both buses isolated; safe-default)
104 *
105 * IMPORTANT: per the user-supplied wiring + this repo's
106 * metadata/e1m_modules/aen/from-cc3501e.tsv, BOTH IO20 and IO21 are
107 * proxied through the on-module CC3501E (GPIO26 and GPIO30 on the
108 * CC3501E side). Firmware drives the mux by dispatching
109 * GPIO_WRITE commands to the CC3501E over the inter-chip SPI1
110 * (see <alp/protocol/cc3501e.h>'s ALP_CC3501E_CMD_GPIO_WRITE),
111 * NOT via Alif's GPIO peripheral.
112 *
113 * EVK_PIN_SDIO_MUX_EN (= ALP_E1M_GPIO_IO20) and EVK_PIN_SDIO_MUX_SEL
114 * (= ALP_E1M_GPIO_IO21) are defined in the generated routes header. */
115
116typedef enum {
117 EVK_SDIO_M2E_KEY = 0,
118 EVK_SDIO_SDCARD = 1,
119} evk_sdio_select_t;

182/* M.2 E-key W_DISABLE control lines (per the M.2 spec).
183 *
184 * Two open-drain signals the host uses to disable the radios on
185 * a populated M.2 E-key card:
186 * W_DISABLE1# -- asserted low, disables Wi-Fi.
187 * W_DISABLE2# -- asserted low, disables Bluetooth.
188 *
189 * Both pads route through the on-module CC3501E (GPIO16 and
190 * GPIO17 per from-cc3501e.tsv). Apps drive them via
191 * ALP_CC3501E_CMD_GPIO_CONFIGURE (with open-drain direction =
192 * input pull-up + write 0 to assert / write 1 / Hi-Z to release)
193 * + ALP_CC3501E_CMD_GPIO_WRITE. The CC3501E firmware needs the
194 * open-drain mode wired into its GPIO_CONFIGURE handler -- a v0.4
195 * item for the embedded firmware/cc3501e/ tree (GPIO proxy group).
196 *
197 * EVK_PIN_W_DISABLE1 (= ALP_E1M_GPIO_IO17) and EVK_PIN_W_DISABLE2
198 * (= ALP_E1M_GPIO_IO16) are defined in the generated routes header. */

200/* PCIe I2C mux + control-IO-expander cluster.
201 *
202 * The EVK splits ALP_E1M_I2C0 between the PCIe M-key and E-key slots
203 * via a downstream mux gated by PCIE0_I2C.EN. The select line
204 * is driven from a SECOND TCAL9538 I/O expander (also on I2C0,
205 * sitting BEFORE the mux so it stays addressable independent of
206 * which PCIe slot the mux is currently routed to). That second
207 * expander also fans out the PCIe-side discrete control signals
208 * (RST / WAKE / CLKREQ for both slots, plus M2E_ALERT).
209 *
210 * Three E1M-side signals control / observe the cluster:
211 * PCIE0_I2C.EN -> E1M IO10 (drives I2C-mux enable)
212 * PCIE_IO_EXP.INT -> E1M IO7 (interrupt from the PCIe expander)
213 * PCIE_IO_EXP.RST -> E1M IO9 (reset for the PCIe expander)
214 *
215 * The PCIe expander is at I2C address 0x71 (A0=1, A1=0 per the
216 * schematic strap). Its alternate part TCA6408ARSVR uses
217 * R112-only / R145-DNP per the schematic note.
218 *
219 * EVK_PIN_PCIE0_I2C_EN (= ALP_E1M_GPIO_IO10), EVK_PIN_PCIE_IOEXP_INT
220 * (= ALP_E1M_GPIO_IO7) and EVK_PIN_PCIE_IOEXP_RST (= ALP_E1M_GPIO_IO9) are
221 * defined in the generated routes header. */

223/* PCIe LANE 2:1 multiplexer cluster (PI3DBS12212AXUAEX x4 +
224 * SY75602BTWL-TR refclk buffer).
225 *
226 * Four PI3DBS12212 muxes split the SoM's PCIe lanes between the
227 * M.2 M-key (x4) and E-key (x1) slots, plus a refclk buffer
228 * forks the single PCIE0_REFCLK to both keys:
229 *
230 * U42 TX0/TX1 mux (E-key gets TX0 only; M-key gets TX0/TX1)
231 * U43 RX0/RX1 mux (mirror for RX)
232 * U50 TX2/TX3 mux (M-key only -- E-key has no lane 2/3)
233 * U51 RX2/RX3 mux (M-key only)
234 * U44 SY75602 (REFCLK fork to both keys)
235 *
236 * The four lane muxes share two control pins:
237 * PCIe.MUX_PD = E1M IO22 (1 = power-down, all outputs Hi-Z)
238 * PCIe.MUX_SEL = E1M IO23 (selects between M-key and E-key)
239 *
240 * Apps that switch slots should:
241 * 1. Power-down the active link first (PD = 1).
242 * 2. Flip SEL.
243 * 3. Wait the PCIe spec's PERST# de-assertion margin.
244 * 4. Bring the muxes back up (PD = 0).
245 * 5. Trigger PCIe re-enumeration on the host.
246 *
247 * U50/U51 take level-shifted versions (`PCIe.MUX_PD_L` and
248 * `_SEL_L`) of the same signals -- U49 (a TXS0102) sits between
249 * the SoM's IO22/IO23 and the PD_L/SEL_L nets. Apps don't need
250 * to think about that; the level shifter is transparent.
251 *
252 * EVK_PIN_PCIE_MUX_PD (= ALP_E1M_GPIO_IO22) and EVK_PIN_PCIE_MUX_SEL
253 * (= ALP_E1M_GPIO_IO23) are defined in the generated routes header. */

391/* ================================================================== */
392/* LCPI 8-bit parallel camera (J22, F32R-1A7H1-11024) */
393/* */
394/* The third camera path on this EVK is a "Low-Power Parallel Camera */
395/* Interface" -- an 8-bit DVP bus on connector J22. Used for */
396/* low-resolution sensors (e.g. Himax HM01B0, OmniVision OV7670) */
397/* that don't support MIPI CSI. The data bus is wired to E1M's */
398/* parallel-camera "D[1:8]" pads -- BIT-REVERSED relative to the */
399/* camera's natural CAM_D[7:0] ordering: */
400/* */
401/* Camera bit E1M pad Camera bit E1M pad */
402/* ---------- ------- ---------- ------- */
403/* CAM_D7 MSB D1 CAM_D3 D5 */
404/* CAM_D6 D2 CAM_D2 D6 */
405/* CAM_D5 D3 CAM_D1 D7 */
406/* CAM_D4 D4 CAM_D0 LSB D8 */
407/* */
408/* Sync / clock lines are NOT bit-reversed: */
409/* CAM_VSYNC -> ALP_E1M_CAM_VSYNC */
410/* CAM_HSYNC -> ALP_E1M_CAM_HSYNC */
411/* CAM_XCLK -> ALP_E1M_CAM_XCLK (host-driven sensor clock) */
412/* CAM_PCLK -> ALP_E1M_CAM_PCLK (sensor pixel clock) */
413/* */
414/* I2C (sensor configuration) is on ALP_E1M_I2C1 (DSI_CSI_I2C, shared */
415/* with the MIPI camera + display panel). */
416/* */
417/* Power: +V_CAM0 (+1V2 nominal) on the connector for analog rails, */
418/* +2V6_CAM for 2.8 V sensor supply. Both are gated by the on-board */
419/* camera LDOs and instrumented by INA236 channels VCAM0 / VCAM1. */
420/* */
421/* Reset / enable signals: */
422/* CAM_RST active-low, on E1M IO5 (alp_gpio_*). */
423/* CAM_EN active-low, on TCAL9538 main P2 (drive LOW to enable). */
424/* */
425/* Apps configure the parallel camera path via the studio's camera */
426/* peripheral binding or by opening alp_camera with the LCPI mode; */
427/* the bit-reversed data wiring is handled inside the SDK's pinmux */
428/* overlay so user code never has to swap bits manually. */
429/* ================================================================== */

559 * CK_RXD -> EVK_UART_PORT_ARDUINO RX
560 * CK_SCL -> EVK_I2C_BUS_ARDUINO SCL (I3C_SCL)
561 * CK_SDA -> EVK_I2C_BUS_ARDUINO SDA (I3C_SDA)
562 *
563 * Sharing means a mikroBUS click + Arduino shield cannot both
564 * drive any of the shared lines simultaneously -- the EVK trusts
565 * the user to mount only one of the two physical headers. */
566
567/* ================================================================== */
568/* On-board sensor 7-bit I2C addresses -- GENERATED, see */
569/* alp_e1m_evk_routes.h */
570/* */
571/* All on ALP_E1M_I2C0 (the sensor bus). Strap values per the EVK */
572/* schematic UG-E1M-001 + user-supplied confirmation; single source */
573/* of truth is the `i2c_devices:` block in */
574/* metadata/boards/e1m-evk.yaml -- do not hand-edit the generated */
575/* macros below: */
576/* - ICM-42670-P (U12) AD0 -> VIO -> 0x69 *** COLLISION, see note *** */
577/* - BMI323 (U13) SDO -> VIO -> 0x69 *** COLLISION, see note *** */
578/* - BMP581 (U14) SDO -> VIO -> 0x47 (SDO must not float) */
579/* - TCAL9538 A1=1, A0=0 -> 0x72 */
580/* ================================================================== */
581
582/* BENCH-CONFIRMED (2026-06-16, E1M-AEN801): U12 (ICM-42670) and U13 (BMI323) BOTH
583 * have their address pins tied to VIO, so both answer at 0x69 and COLLIDE on the bus
584 * -- each ACKs with a different read framing (ICM 8-bit direct vs BMI323 2-dummy-byte),
585 * so a read returns wired-AND garbage (seen: 0x0001) and neither IMU is individually
586 * addressable. Nothing answers at 0x68. The next batch must re-strap ONE of them -- e.g.
587 * tie the BMI323 SDO pin to GND -> 0x68 (its datasheet default). The macros carry
588 * that de-conflicted target so firmware is correct post-respin; on the pre-respin batch
589 * the IMUs cannot be read (HW conflict, not a driver bug).
590 *
591 * EVK_I2C_ADDR_ICM42670, EVK_I2C_ADDR_BMI323 and EVK_I2C_ADDR_BMP581 are
592 * defined in the generated routes header. */
593
594/* BMI323 INT1 (data-ready / motion / FIFO interrupt) routes to
595 * E1M IO15, NOT through the main TCAL9538 expander (the expander's
596 * P4..P7 hold the ICM-42670 + BMP581 interrupts).
597 *
598 * IO15 is a CC3501E-side pad on this SoM (CC3501E GPIO14 per
599 * metadata/e1m_modules/aen/from-cc3501e.tsv) -- apps must register
600 * an interrupt via ALP_CC3501E_CMD_GPIO_SET_INTERRUPT and receive
601 * the rising/falling edge through the CC3501E event callback,
602 * rather than installing an alp_gpio handler against Alif's GPIO
603 * peripheral.
604 *
605 * EVK_PIN_BMI323_INT1 (= ALP_E1M_GPIO_IO15) is defined in the generated
606 * routes header. */
607
608/* The EVK populates TWO TCAL9538 I/O expanders, both on ALP_E1M_I2C0
609 * but at different strap-selected addresses:
610 * - The "main" expander handles LCD / camera / capacitive-touch
611 * control + four sensor interrupt inputs (see
612 * evk_ioexp_pin_t). Strap A1=1, A0=0 -> 0x72.
613 * - The "PCIe" expander handles the I2C-mux SEL + PCIe slot
614 * RST/WAKE/CLKREQ signals + M2E_ALERT (see
615 * evk_pcie_ioexp_pin_t above). Strap A0=1, A1=0 -> 0x71.
616 *
617 * EVK_I2C_ADDR_TCAL9538_MAIN and EVK_I2C_ADDR_TCAL9538_PCIE are defined
618 * in the generated routes header. */
619
620/* BENCH-CONFIRMED (2026-06-16): U35 can be assembled with the TCA6408ARSVR
621 * alternative (R112 fitted, R145 DNP) instead of the TCAL9538, which moves it to
622 * 0x20. It is PCA9538-register-compatible (0x00 input / 0x01 output / 0x02 polarity
623 * / 0x03 config), so the chips/tcal9538 driver drives it unchanged at 0x20 -- read
624 * back config=0xFF + a live input port on the bench.
625 *
626 * EVK_I2C_ADDR_TCA6408A_MAIN (the TCA6408A-populated variant) and
627 * EVK_I2C_ADDR_TCAL9538 (convenience alias for EVK_I2C_ADDR_TCAL9538_MAIN)
628 * are defined in the generated routes header. */
629
630/* Two TAS2563RPP smart-amp ICs share the same I2C0 bus. AD0
631 * strap selects address per TAS2563 datasheet table 7-3:
632 * AD0 = 10k to GND -> 0x4D -- U27 on the EVK
633 * AD0 = 10k to VDD -> 0x4E -- U28 on the EVK
634 * The TAS2563 also has a hardware broadcast page-write convention at
635 * 0x48. On PRE-RESPIN boards that address was occupied by U32
636 * INA236B (+V_CAM0 rail), so the broadcast was unusable; U32 was
637 * re-strapped A0=SCL -> 0x4B from the next batch, which frees 0x48.
638 * Firmware that must work on both board revisions still has to issue
639 * two targeted unit-address writes back-to-back rather than relying
640 * on a 0x48 broadcast.
641 *
642 * EVK_I2C_ADDR_TAS2563_LOW and EVK_I2C_ADDR_TAS2563_HIGH are defined
643 * in the generated routes header. */
644
645/* Six INA236 high-side current-shunt monitors -- one per power
646 * rail -- on I2C0. TI's INA236A variant occupies 0x40..0x43 and
647 * the INA236B variant occupies 0x48..0x4B, so all six fit on one
648 * bus despite each variant only having 2 strap bits.
649 *
650 * EVK ref-des per the user-confirmed schematic:
651 * U21 INA236A 3V3 rail A0 = GND -> 0x40
652 * U31 INA236A 1V8 rail A0 = V+ -> 0x41
653 * U33 INA236A VIO rail A0 = SDA -> 0x42
654 * U32 INA236B +V_CAM0 A0 = SCL -> 0x4B (re-strapped from the pre-respin
655 * A0=GND -> 0x48, which collided
656 * with the TAS2563 broadcast address)
657 * U34 INA236B +V_CAM1 A0 = V+ -> 0x49
658 * U30 INA236B +5V rail A0 = SDA -> 0x4A
659 *
660 * INA236A variants occupy 0x40..0x43 (A0 strap = GND/VS/SDA/SCL);
661 * INA236B variants occupy 0x48..0x4B with the same A0 encoding.
662 * Confirmed against the EVK schematic strap labels. The three
663 * B-bank addresses actually in use (0x49 / 0x4A / 0x4B) collide with
664 * nothing: not the TAS2563 unit addresses (0x4D / 0x4E), and not its
665 * broadcast address (0x48), which the U32 re-strap freed. On
666 * PRE-RESPIN boards U32 sat at 0x48 and did shadow that broadcast --
667 * see the TAS2563 block above for the two-write workaround that
668 * remains necessary if you support both revisions.
669 *
670 * EVK_I2C_ADDR_INA236_3V3, _1V8, _VIO, _VCAM0, _VCAM1 and _5V are
671 * defined in the generated routes header. */
672
673/* Per-rail shunt + max-current values for ina236_init(). All six
674 * EVK rails picked their shunt to put the rail's nominal max
675 * current near the INA236's 81.92 mV full-scale shunt voltage:
676 * shunt_ohms * max_current_a ~= 0.080 V.
677 * Apps can pass these directly:
678 * ina236_init(&ctx, bus,
679 * EVK_I2C_ADDR_INA236_3V3,
680 * EVK_INA236_SHUNT_3V3_OHMS,
681 * EVK_INA236_MAX_3V3_A,
682 * INA236_ADCRANGE_81MV);
683 *
684 * EVK_INA236_SHUNT_<RAIL>_OHMS and EVK_INA236_MAX_<RAIL>_A (RAIL =
685 * 3V3, 1V8, VIO, VCAM0, VCAM1, 5V) are defined in the generated
686 * routes header. */

688/* ================================================================== */
689/* On-board audio I/O */
690/* */
691/* Audio in (PDM): */
692/* Four STMicro MP34DT05TR-A digital MEMS microphones, two pairs: */
693/* U19 LEFT + U20 RIGHT on E1M PDM0 (CK = PDM_C0, D = PDM_D0) */
694/* U25 LEFT + U26 RIGHT on E1M PDM1 (CK = PDM_C1, D = PDM_D1) */
695/* Each LEFT/RIGHT pair shares one PDM data line -- LEFT mics tie */
696/* their LR pin high (data on rising clock edge), RIGHT mics tie */
697/* LR low (data on falling edge), and the single data line carries */
698/* both interleaved. Apps open the streams via alp_audio_in_open */
699/* with peripheral_id = 0 (PDM0) or 1 (PDM1), channels = 2 (stereo). */
700/* */
701/* Audio out (I2S): */
702/* Two TAS2563RPP smart-amp ICs (one channel each, addresses 0x4D */
703/* and 0x4E) sit on E1M I2S0. Audio flows host -> amp on */
704/* I2S0_SDO; the amp reports IV-sense data back on I2S0_SDI. */
705/* Use chips/tas2563 to bring them up; AMP.ENABLE / AMP.FAULT are */
706/* on EVK_PIN_AMP_ENABLE / EVK_PIN_AMP_FAULT respectively. */
707/* */
708/* I2S0 itself can be re-routed away from the amps to the M.2 */
709/* E-key slot via the I2S 74LVC157 mux -- see */
710/* EVK_PIN_I2S_MUX_EN + EVK_PIN_I2S_MUX_SEL above. */
711/* ================================================================== */
