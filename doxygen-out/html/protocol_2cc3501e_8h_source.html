180 /* OTA firmware update (over-the-bridge). The Alif host streams a signed
181 * GPE-format vendor image into the CC3501E's NON-primary vendor slot via
182 * PSA-FWU; on FINISH the CC35 installs + reboots so the cold BL2/MCUboot
183 * swaps the slot to primary (TRIAL boot), and the swapped image accepts
184 * itself (cc3501e_hw_tick). Streamed sequentially: BEGIN(total_len) ->
185 * WRITE(offset,bytes)* -> FINISH. See docs/cc3501e-bridge.md "OTA" + the
186 * device-side Mender contract (the OTA server is a separate repo). */
187 ALP_CC3501E_CMD_OTA_BEGIN = 0x40, /* req alp_cc3501e_ota_begin_t */
188 ALP_CC3501E_CMD_OTA_WRITE = 0x41, /* req alp_cc3501e_ota_write_t + bytes */
189 ALP_CC3501E_CMD_OTA_FINISH = 0x42, /* no payload; install + deferred reboot */
190 ALP_CC3501E_CMD_OTA_ABORT = 0x43, /* no payload; cancel the session */
191 ALP_CC3501E_CMD_OTA_STATUS = 0x44, /* reply alp_cc3501e_ota_status_t */
192 /* 0x45 is STREAM_WRITE (below), so OTA_PROMOTE takes the next free code.
193 * Requests the swap-reboot for an image ALREADY committed to STAGED (e.g.

195 * unjam/promote path FINISH cannot re-reach once a slot is occupied. */

803/* ------------------------------------------------------------------ */
804/* Wire-layout guards (#733) */
805/* */
806/* Two categories of struct live in this header: */
807/* */
808/* (A) DIRECTLY SERIALIZED -- the host builds the struct and hands */
809/* its raw address to the SPI DMA, and the firmware casts the */
810/* received wire buffer straight back to the struct type. Its */
811/* in-memory layout IS the wire format, so ANY tail/interior */
812/* padding silently corrupts the frame. These are padding-free */
813/* TODAY only because every field is uint8_t; adding a wider */
814/* field would reintroduce the alp_cc3501e_ble_adv_start_t */
815/* 7-vs-8 gap. The asserts below make that a BUILD failure */
816/* instead of a field-shift on the wire -- keep these all- */
817/* uint8_t, or switch the op to a hand-packed codec before */
818/* widening a field. */
819/* */
820/* (B) WIRE-SCHEMA -- documentary layout only. Both sides encode/ */
821/* decode these field-by-field with explicit little-endian byte */
822/* access (never memcpy / pointer-cast), so compiler padding is */
823/* irrelevant to the wire. Where the firmware still uses */
824/* sizeof(T) as the expected payload length (the socket ops), */
825/* the size is pinned below so a field reorder can't drift that */
826/* length constant out from under the byte-offset parser. */
827/* ------------------------------------------------------------------ */
828
829/* (A) Directly-serialized command payloads -- struct-punned on both */
830/* sides (chips/cc3501e/cc3501e.c hands &struct to the SPI DMA; */
831/* firmware/cc3501e/src/protocol.c casts the wire buffer to T*). */
832_Static_assert(sizeof(alp_cc3501e_gpio_configure_t) == 4u,
833 "gpio_configure wire header must stay 4 bytes / padding-free");
834_Static_assert(offsetof(alp_cc3501e_gpio_configure_t, cc3501e_gpio) == 0u, "gpio_configure @0");
835_Static_assert(offsetof(alp_cc3501e_gpio_configure_t, direction) == 1u,
836 "gpio_configure.direction @1");
837_Static_assert(offsetof(alp_cc3501e_gpio_configure_t, pull) == 2u, "gpio_configure.pull @2");
838
839_Static_assert(sizeof(alp_cc3501e_gpio_write_t) == 4u,
840 "gpio_write wire header must stay 4 bytes / padding-free");
