58/* Five INA236 monitors on XEVK_I2C_BUS_SENSORS. Ref-des + rail +
59 * A0 strap per the board schematic (Current Measurement.SchDoc):
60 *
61 * U21 INA236A 3V3 rail A0 = GND -> 0x40 (20 mOhm)
62 * U31 INA236A 1V8 rail A0 = V+ -> 0x41 (20 mOhm)
63 * U32 INA236B VCAM2 rail A0 = GND -> 0x48 (50 mOhm)
64 * U34 INA236B VCAM3 rail A0 = V+ -> 0x49 (50 mOhm)
65 * U30 INA236B 5V rail A0 = SDA -> 0x4A (20 mOhm)
66 *
67 * INA236A occupies 0x40..0x43, INA236B occupies 0x48..0x4B (same
68 * A0 strap encoding), so all five share the one bus.
69 *
70 * NEXT-REVISION board notes (observed on current silicon, 2026-06;
71 * NOT exposed as macros -- do not rely on them):
72 * - 0x42 / 0x43 also ACK as INA236 (mfg-ID "TI") although the
73 * schematic BOM lists only the five monitors above; treated as
74 * a board anomaly, to be resolved on the next respin.
75 * - The 3V3 (U21) and 1V8 (U31) monitors read ~0 V on the bus-
76 * voltage register on current silicon (VBUS-sense wiring under
77 * investigation); their shunt/current path is unaffected. 5V
78 * (U30) reads correctly (~4.88 V / whole-board input current).
79 */
80#define XEVK_I2C_ADDR_INA236_3V3 0x40u
81#define XEVK_I2C_ADDR_INA236_1V8 0x41u
82#define XEVK_I2C_ADDR_INA236_VCAM2 \
83 0x48u
84#define XEVK_I2C_ADDR_INA236_VCAM3 \
