231 /* v0.4: GD32G5 TMU (CORDIC) math accelerator. Standalone block
232 * (NOT an ADC postprocessor): sin/cos/tan, atan/atan2, sqrt, log,

234 * request payload; reply carries `result:u32 status:u8`. Format

236 GD32G553_CMD_TMU_COMPUTE = 0x90,
237 /* v0.5: ADC-stream DSP pipeline configuration -- RESERVED +
238 * tombstoned. Original single-shot configure can't fit a FIR
239 * stage's 256-byte Q31-tap blob in the 65-byte wire envelope, so
240 * the actual upload path is the three chunked sub-opcodes
241 * CMD_ADC_DSP_CHAIN_* (0x37/0x38/0x39) immediately below. The
242 * 0x36 opcode keeps its slot to avoid renumbering across the
243 * v0.5.x line; firmware default-case dispatch returns
244 * STATUS_NOSUPPORT for it. */
245 GD32G553_CMD_ADC_STREAM_CONFIGURE_DSP = 0x36,
246 /* v0.5 (§2B wave-2): chunked DSP-chain upload. CHAIN_OPEN
247 * allocates a firmware-side chain handle and replies with the
248 * assigned `chain_id:u8`; STAGE_PUSH uploads one chunk of one
249 * stage's per-kind params (FIR taps / IIR sections / WINDOW shape
250 * / FFT size + output format) by `chain_id:u8 stage_index:u8
251 * kind:u8 chunk_offset:u16 chunk_total_size:u16 chunk_data[...]`,
252 * repeated until the per-kind blob is fully assembled; CHAIN_BIND
253 * attaches the completed chain to a streaming ADC source by
254 * `chain_id:u8 stream_id:u8` -- after which the stream's samples
255 * flow through the chain instead of straight to the host. All
256 * three opcodes are RESERVED at protocol v0.5 -- firmware default-
257 * case dispatch returns STATUS_NOSUPPORT until the
258 * `bridge_hw_adc_dsp_*` HAL bodies land in the GD32 firmware
259 * tree. Host helpers in `chips/gd32g553/` honour the same
260 * NOSUPPORT contract by routing through cmd_send unchanged.
261 * See `docs/gd32-bridge-protocol.md` §3.x for the wire layout. */

263 GD32G553_CMD_ADC_DSP_STAGE_PUSH = 0x38,
264 GD32G553_CMD_ADC_DSP_CHAIN_BIND = 0x39,
265 GD32G553_CMD_ADC_SPECTRUM_READ = 0x3A,
266 /* v0.5 (§2B.2): advanced timer extras. PWM_CAPTURE turns a PWM
267 * channel's pin into an input-capture source for frequency / pulse-
268 * width measurement; PWM_SINGLE_PULSE drives a one-shot pulse of
269 * caller-specified duration then stops; TIMER_SYNC links the
270 * GD32G5's TIMER0 / TIMER7 / TIMER19 in master-slave configuration.
271 * All five opcodes are RESERVED at protocol v0.5; the firmware
272 * default-case path returns STATUS_NOSUPPORT until the corresponding
273 * bridge_hw_* HAL bodies land. Portable surfaces in <alp/pwm.h> +
274 * <alp/counter.h> mirror the same NOSUPPORT contract on builds

1014/* ------------------------------------------------------------------ */
1015/* OTA -- in-system upgrade of the bridge firmware */
1016/* */
1017/* Opcodes 0xF0..0xF6 implement the Path-A application-bootloader */
1018/* upgrade (docs/gd32-bridge-protocol.md §10): BEGIN announces */
1019/* size+CRC and erases the inactive slot, WRITE_CHUNK streams the */
1020/* image, VERIFY CRC-checks it, COMMIT flips the A/B metadata and */
1021/* resets into the new slot, ROLLBACK flips back. The firmware side */
1022/* is SAFE-BY-DEFAULT: only a -DBRIDGE_OTA_PARTITIONED build (paired */
1023/* with the bootloader + slot-linked apps) arms the flash path; */
1024/* default builds answer STATUS_NOSUPPORT to the whole range, which */
1025/* these helpers surface as ALP_ERR_NOSUPPORT. */
1026/* */
1027/* Transaction note: BEGIN (slot erase), VERIFY (full-image CRC) and */
1028/* COMMIT/ROLLBACK (reset before the reply is drained) block the */
1029/* bridge long enough that the reply transaction can miss -- treat */
1030/* ALP_ERR_IO from those as "issued, confirm via GET_STATE" (or, for */
1031/* COMMIT/ROLLBACK, by re-initialising against the rebooted bridge). */
1032/* ------------------------------------------------------------------ */
